cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	project(wacca_mercuryio LANGUAGES C)
	set(CMAKE_C_STANDARD 23)
	set(CMAKE_C_STANDARD_REQUIRED ON)
	set(CMAKE_C_EXTENSIONS OFF)
	set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
	set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
	enable_testing()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
	message(FATAL_ERROR "mercuryio.dll requires a 64-bit Windows compiler for WACCA.")
endif()

set(MERCURY_COMMON "${CMAKE_CURRENT_SOURCE_DIR}/../firmware/common")
add_library(mercury_wire STATIC "${MERCURY_COMMON}/protocol.c")
target_include_directories(mercury_wire PUBLIC "${MERCURY_COMMON}")
add_library(mercuryio SHARED
	src/mercuryio.c src/config.c src/frame.c src/serial.c mercuryio.def)
target_include_directories(mercuryio PRIVATE include src)
target_link_libraries(mercuryio PRIVATE mercury_wire user32)
target_compile_definitions(mercuryio PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
set_target_properties(mercuryio PROPERTIES PREFIX "" OUTPUT_NAME "mercuryio")

add_executable(mercury_frame_test tests/test_frame.c src/frame.c src/config.c)
target_include_directories(mercury_frame_test PRIVATE include src)
target_link_libraries(mercury_frame_test PRIVATE mercury_wire)
add_test(NAME mercury_frame COMMAND mercury_frame_test)

add_executable(mercury_serial_test tests/test_serial.c src/frame.c src/config.c)
target_include_directories(mercury_serial_test PRIVATE include src)
target_link_libraries(mercury_serial_test PRIVATE mercury_wire)
target_compile_definitions(mercury_serial_test PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
add_test(NAME mercury_serial COMMAND mercury_serial_test)

add_executable(mercury_abi_test tests/test_abi.c)
target_include_directories(mercury_abi_test PRIVATE include)
target_compile_definitions(mercury_abi_test PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
add_dependencies(mercury_abi_test mercuryio)
add_test(NAME mercury_abi COMMAND mercury_abi_test "$<TARGET_FILE:mercuryio>")

add_executable(mercury_worker_test tests/test_worker.c src/mercuryio.c src/frame.c)
target_include_directories(mercury_worker_test PRIVATE include src)
target_link_libraries(mercury_worker_test PRIVATE mercury_wire user32)
target_compile_definitions(mercury_worker_test PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
add_test(NAME mercury_worker COMMAND mercury_worker_test)
set_tests_properties(mercury_worker PROPERTIES TIMEOUT 15)

foreach(target mercury_wire mercuryio mercury_frame_test mercury_serial_test mercury_abi_test mercury_worker_test)
	if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
		target_compile_options(${target} PRIVATE
			-Wall -Wextra -Werror -Wconversion -Wshadow -Wstrict-prototypes -Wpedantic)
	elseif(MSVC)
		target_compile_options(${target} PRIVATE /W4 /WX)
	endif()
endforeach()
foreach(target mercury_frame_test mercury_serial_test mercury_abi_test mercury_worker_test)
	if(MSVC)
		target_compile_options(${target} PRIVATE /UNDEBUG)
	else()
		target_compile_options(${target} PRIVATE -UNDEBUG)
	endif()
endforeach()
if(MINGW)
	target_link_options(mercuryio PRIVATE -static-libgcc -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
# Do not overwrite a user's configured mercuryio.ini during a rebuild.
configure_file(mercuryio.example.ini "${CMAKE_BINARY_DIR}/mercuryio.example.ini" COPYONLY)
